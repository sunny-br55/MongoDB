// insert the data
db.products.insertMany(products)

db.products.countDocuments()

// price index
db.products.find({
  price: { $gt: 50000 }
}).explain("executionStats")

db.products.createIndex({ price: 1 })

db.products.find({
  price: { $gt: 50000 }
}).explain("executionStats")

// compound index
db.products.find({
  category: "Laptop",
  price: { $gt: 50000 }
}).explain("executionStats")

db.products.createIndex({
  category: 1,
  price: -1
})

db.products.find({
  category: "Laptop",
  price: { $gt: 50000 }
}).explain("executionStats")

// index on tags
db.products.createIndex({ tags: 1 })

db.products.find({
  tags: "student"
}).explain("executionStats")

// text search
db.products.createIndex({
  description: "text"
})

db.products.find({
  $text: { $search: "laptop" }
}).explain("executionStats")

// hashed index
db.products.createIndex({
  productId: "hashed"
})

db.products.find({
  productId: 5
})

// location index
db.products.createIndex({
  location: "2dsphere"
})

db.products.find({
  location: {
    $near: {
      $geometry: {
        type: "Point",
        coordinates: [75.80, 26.90]
      },
      $maxDistance: 10000
    }
  }
})

// unique sparse index
db.products.createIndex(
  { email: 1 },
  {
    unique: true,
    sparse: true
  }
)

db.products.findOne({
  email: "rahul1@example.com"
})

// partial index
db.products.createIndex(
  { price: 1 },
  {
    partialFilterExpression: {
      status: "active"
    }
  }
)

// wildcard index
db.products.createIndex({
  "$**": 1
})

// show indexes
db.products.getIndexes()
