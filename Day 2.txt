// and

// or

    $or: [
        { department: "CSE" },
        { department: "IT" }
    ]
})

// not
db.students.find({
    marks: {
        $not: {
            $gt: 80
        }
    }
})

// nor
db.students.find({
    $nor: [
        { department: "CSE" },
        { department: "IT" }
    ]
})


// equal

    }
})

// greater than

    }
})

// less than

    marks: {
        $lt: 70
    }
})

// greater than or equal
db.students.find({
    marks: {
        $gte: 85
    }
})

// less than or equal
db.students.find({
    marks: {
        $lte: 70
    }
})

// not equal

    department: {
        $ne: "CSE"
    }
})

    department: {

    }

        }

    }

    }
