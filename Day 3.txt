use collegeDB

// delete previous data
db.shopOrders.drop()

// data
const itemTypes = [
    "Notebook", "Pen", "Backpack", "Water Bottle", "Calculator",
    "Earphones", "Mouse", "Keyboard", "USB Drive", "Desk Lamp"
]

const departments = [
    "CSE", "ECE", "ME", "Civil", "AI", "IT"
]

const brands = [
    "Classmate", "Cello", "Wildcraft", "Casio",
    "JBL", "Logitech", "HP", "Portronics"
]

const cities = [
    "Jaipur", "Delhi", "Pune", "Lucknow",
    "Patna", "Bhopal", "Indore", "Kota"
]

const payment = [
    "UPI", "Card", "Cash", "Net Banking"
]

const status = [
    "Delivered", "Shipped", "Pending", "Cancelled", "Returned"
]

const sellers = [
    "CampusStore", "TechPoint", "BookCorner",
    "DailyNeeds", "StudentMart"
]

const colours = [
    "Black", "Blue", "Red", "White", "Green"
]

const tags = [
    "college", "daily-use", "new", "offer",
    "popular", "useful"
]

let data = []

// inserting records
for (let i = 1; i <= 3000; i++) {

    const item = itemTypes[Math.floor(Math.random() * itemTypes.length)]
    const department = departments[Math.floor(Math.random() * departments.length)]
    const brand = brands[Math.floor(Math.random() * brands.length)]
    const city = cities[Math.floor(Math.random() * cities.length)]

    const paymentMode =
        payment[Math.floor(Math.random() * payment.length)]

    const orderStatus =
        status[Math.floor(Math.random() * status.length)]

    const seller =
        sellers[Math.floor(Math.random() * sellers.length)]

    const quantity = Math.floor(Math.random() * 5) + 1
    const price = Math.floor(Math.random() * 4900) + 100
    const discount = Math.floor(Math.random() * 31)

    const finalPrice = Math.round(
        price - (price * discount / 100)
    )

    const rating = Number(
        (Math.random() * 4 + 1).toFixed(1)
    )

    const stock = Math.floor(Math.random() * 150)

    const orderDate = new Date(
        2025,
        Math.floor(Math.random() * 12),
        Math.floor(Math.random() * 28) + 1
    )

    const itemTags = [
        tags[Math.floor(Math.random() * tags.length)],
        tags[Math.floor(Math.random() * tags.length)]
    ]

    const order = {

        orderId: "ORD" + String(i).padStart(4, "0"),

        itemName: item + " " + i,

        itemType: item,

        department: department,

        brand: brand,

        price: price,

        discount: discount,

        finalPrice: finalPrice,

        quantity: quantity,

        amount: finalPrice * quantity,

        rating: rating,

        stock: stock,

        available: stock > 0,

        seller: {
            name: seller,
            rating: Number(
                (Math.random() * 4 + 1).toFixed(1)
            )
        },

        customer: {
            name: "Student" + (Math.floor(Math.random() * 500) + 1),
            city: city,
            age: Math.floor(Math.random() * 23) + 18
        },

        payment: {
            mode: paymentMode,
            transactionId:
                "TRX" + Math.random().toString(36).substring(2, 10)
        },

        orderStatus: orderStatus,

        orderDate: orderDate,

        tags: itemTags,

        details: {
            colour: colours[
                Math.floor(Math.random() * colours.length)
            ],
            warranty:
                (Math.floor(Math.random() * 3) + 1) + " year",
            weight:
                Number((Math.random() * 3 + 0.2).toFixed(2))
        },

        featured: Math.random() > 0.75,

        createdAt: new Date()
    }

    data.push(order)

    // insert 500 at a time
    if (data.length === 500) {
        db.shopOrders.insertMany(data)
        data = []
    }
}

if (data.length > 0) {
    db.shopOrders.insertMany(data)
}

print("3000 records inserted")

    {
        $project: {
            tags: {

                    input: "$tags",

                }
            }
        }
    }
])
