db.students.dropIndex("name_1")

// Single-field index
db.students.createIndex({ rollNo: 1 })
db.students.find({ rollNo: 101 })

// Compound index
db.students.createIndex({ department: 1, marks: -1 })
db.students.find({ department: "CSE" }).sort({ marks: -1 })

// Multikey index
db.students.insertOne({
  name: "Aman",
  skills: ["Java", "MongoDB", "Python"]
})
db.students.createIndex({ skills: 1 })
db.students.find({ skills: "MongoDB" })

// Explain plan
db.students.find({ rollNo: 101 }).explain("executionStats")

// Hidden index
db.students.createIndex({ email: 1 })
db.students.hideIndex("email_1")
db.students.unhideIndex("email_1")
db.students.getIndexes()

// Unique index
db.students.createIndex({ email: 1 }, { unique: true })

// Index validation
db.students.find({ rollNo: 101 }).explain("executionStats")
